function(idol_add_catch_executable target source)
    add_executable(${target} ${source})
    target_link_libraries(${target} PRIVATE idol Catch2::Catch2WithMain)
endfunction()

function(idol_register_catch_test name target filter labels)
    if (filter STREQUAL "")
        add_test(NAME ${name} COMMAND ${target})
    else ()
        add_test(NAME ${name} COMMAND ${target} ${filter})
    endif ()
    set_tests_properties(${name} PROPERTIES LABELS "${labels}" TIMEOUT 60)
endfunction()

function(idol_solver_labels solver output)
    set(labels "${solver}")
    if (solver STREQUAL "GLPK" OR solver STREQUAL "HiGHS" OR solver STREQUAL "OsiClp")
        list(APPEND labels free-solver)
    elseif (solver STREQUAL "Gurobi" OR solver STREQUAL "Cplex")
        list(APPEND labels commercial-solver)
    endif ()
    set(${output} "${labels}" PARENT_SCOPE)
endfunction()
